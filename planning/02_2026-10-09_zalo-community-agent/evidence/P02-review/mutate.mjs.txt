const id = process.argv[2];
const m = M[id];
if (!m) { console.error('unknown mutation', id); process.exit(2); }
const src = readFileSync(m.file, 'utf8');
const n = src.split(m.from).length - 1;
if (n !== 1) { console.log(`${id}: APPLY_FAILED (matches=${n})`); process.exit(3); }
writeFileSync(m.file, src.replace(m.from, m.to));
const lines = [`${id}: ${m.file}`, `  - ${JSON.stringify(m.from)}`, `  + ${JSON.stringify(m.to)}`];
try {
  for (const r of m.runs) {
    const p = spawnSync(r.cmd, { cwd: r.cwd, shell: true, encoding: 'utf8', env: process.env, maxBuffer: 64 * 1024 * 1024 });
    const out = (p.stdout + p.stderr).replace(/\x1b\[[0-9;]*m/g, '');
    const summary = out.split('\n').filter((l) => /Tests\s+\d|FAIL |forbidden_|unlisted_|AssertionError|expected/.test(l)).slice(0, 12);
    lines.push(`  [${r.cwd}] ${r.cmd} ⇒ exit ${p.status}`);
    for (const s of summary) lines.push(`      ${s.trim().slice(0, 220)}`);
  }
} finally {
  execSync(`git checkout -- "${m.file}"`);
  lines.push(`  restored: git checkout -- ${m.file}`);
}
console.log(lines.join('\n'));
