const ACC = 'community';

async function approval(group: string, status: string): Promise<string> {
  const r = await sql(
    `insert into approvals (zalo_account_id, kind, thread_id, target_uid, reason, requested_by, status, expires_at)
     values ($1,'kick_member',$2,'u-victim','probe','agent',$3, now() + interval '1 hour') returning id`,
    [ACC, group, status]
  );
  return r[0].id;
}
const pacer = (s: any, now?: () => number) => new AccountPacer(ACC, H.FAST, () => s.getSession(), { rng: () => 0, ...(now ? { now } : {}) });
const kinds = (fake: any) => fake.actions.map((a: any) => a.kind).join(',') || '(none)';

await H.resetAccount();

// ---- P1 (c): kick/block KHONG qua approvals
{
  const { s, fake } = await H.connectedSession();
  const g = H.uid('pg');
  await H.makeGroup(g);
  const pend = await approval(g, 'pending');
  const rej = await approval(g, 'rejected');
  const id1 = await H.enqueueId({ zaloAccountId: ACC, threadId: g, threadType: 'group', kind: 'action', body: { action: 'remove_member', uid: 'u-victim' } });
  const id2 = await H.enqueueId({ zaloAccountId: ACC, threadId: g, threadType: 'group', kind: 'action', body: { action: 'remove_member', uid: 'u-victim', approvalId: pend } });
  const id3 = await H.enqueueId({ zaloAccountId: ACC, threadId: g, threadType: 'group', kind: 'action', body: { action: 'block_member', uid: 'u-victim', approvalId: rej } });
  await pacer(s).tick();
  const st = await sql(`select id, status from approvals where id = any($1::uuid[])`, [[pend, rej]]);
  log(`P1 zalo calls: ${kinds(fake)}`);
  for (const id of [id1, id2, id3]) log(`P1 outbox ${id}: status=${(await H.outboxRow(id)).status}`);
  log(`P1 approvals after: ${st.map((r: any) => `${r.id === pend ? 'pending-one' : 'rejected-one'}=${r.status}`).join(' ')}`);
  await sql(`delete from approvals where id = any($1::uuid[])`, [[pend, rej]]);
  await H.dropGroup(g);
  await s.stop();
}

// ---- P3 (a): cliMsgId doan = msgId — worker + sim
{
  const { s, fake } = await H.connectedSession();

  const id = await H.enqueueId({ zaloAccountId: ACC, threadId: g, threadType: 'group', kind: 'action', body: { action: 'delete_message', msgId: '777', cliMsgId: '777', uidFrom: 'u1' } });

    `insert into group_messages (zalo_account_id, thread_id, zalo_msg_id, cli_msg_id, sender_uid, text) values ($1,$2,'888','real-cli','u1','x')`,
    [ACC, g]
  );

  await H.dropGroup(g);

// ---- P4 (b): dong ho worker lech DB +61 phut ⇒ ngan sach "reset"
{

      `insert into outbox (zalo_account_id, thread_id, thread_type, kind, body, status, sent_at) values ($1,$2,'group','text','{"text":"x"}','sent', now() - interval '5 minutes')`,
      [ACC, g]
    );
  }
  log(`P4 groupTextSentLastHour(now)=${await store.groupTextSentLastHour(ACC, g, Date.now())} vs (now+61m)=${await store.groupTextSentLastHour(ACC, g, Date.now() + 61 * 60_000)}`);

  const skew = 61 * 60_000;
  const p = pacer(s, () => Date.now() + skew);

  await p.tick();
