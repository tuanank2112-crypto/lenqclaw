const series = new Map(); // name -> {ms:[], errors:[], meta:[]}
function rec(name, ms, meta = {}) {
  if (!series.has(name)) series.set(name, { ms: [], errors: [], meta: [] });
  const s = series.get(name);
  if (meta.error) s.errors.push(meta.error); else s.ms.push(ms);
  s.meta.push(meta);
}
function pct(arr, p) {
  if (arr.length === 0) return NaN;
  const a = [...arr].sort((x, y) => x - y);
  return a[Math.min(a.length - 1, Math.ceil((p / 100) * a.length) - 1)];
}

async function timed(fn) {
  const t0 = performance.now();
  try { const r = await fn(); return { ms: performance.now() - t0, ...r }; }
  catch (e) { return { ms: performance.now() - t0, error: scrub(e?.message ?? e).slice(0, 160) }; }
}
async function gemini(model, body) {
  if (calls >= MAX_CALLS) throw new Error('MAX_CALLS');
  calls++;
  const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(model)}:generateContent`, {
    method: 'POST', headers: { 'content-type': 'application/json', 'x-goog-api-key': GEMINI_KEY }, body: JSON.stringify(body), signal: AbortSignal.timeout(60_000),
  });
  const j = await res.json().catch(() => null);
  if (!res.ok) {
    const q = (j?.error?.details ?? []).flatMap((d) => d.violations ?? []).map((v) => v.quotaId).filter(Boolean).join(',');
    throw new Error(`HTTP ${res.status} quotaId=${q || '-'} ${JSON.stringify(j?.error?.message ?? '').slice(0, 60)}`);
  }
  const c = j?.candidates?.[0];
  const parts = c?.content?.parts ?? [];
  const u = j?.usageMetadata ?? {};
  return { j, content: c?.content, finish: c?.finishReason, fc: parts.find((p) => p.functionCall)?.functionCall?.name ?? null,
    inTok: u.promptTokenCount ?? 0, outTok: u.candidatesTokenCount ?? 0, thinkTok: u.thoughtsTokenCount ?? 0 };
}
async function vyce(model, body) {
  if (calls >= MAX_CALLS) throw new Error('MAX_CALLS');
  calls++;
  const res = await fetch(VYCE_URL, { method: 'POST', headers: { 'content-type': 'application/json', authorization: `Bearer ${VYCE_KEY}` }, body: JSON.stringify({ model, ...body }), signal: AbortSignal.timeout(60_000) });
  const j = await res.json().catch(() => null);
  if (!res.ok) throw new Error(`HTTP ${res.status} ${JSON.stringify(j?.error?.message ?? '').slice(0, 120)}`);
  const ch = j?.choices?.[0];
  return { j, msg: ch?.message, finish: ch?.finish_reason, fc: ch?.message?.tool_calls?.[0]?.function?.name ?? null,
    inTok: j?.usage?.prompt_tokens ?? 0, outTok: j?.usage?.completion_tokens ?? 0, thinkTok: j?.usage?.completion_tokens_details?.reasoning_tokens ?? 0 };
}

async function turnGemini(model, lvl, focus, maxOut) {
  const contents = [{ role: 'user', parts: [{ text: userBlock(focus) }] }];

  if (c1.error || c1.fc !== 'search_knowledge') return { c1, c2: null };
  const callPart = c1.content.parts.find((p) => p.functionCall);
  const contents2 = [...contents, c1.content, { role: 'user', parts: [{ functionResponse: { name: 'search_knowledge', response: TOOL_RESULT } }] }];
  void callPart;
  const c2 = await timed(() => gemini(model, { systemInstruction: { parts: [{ text: SYSTEM_BIG }] }, contents: contents2, tools: geminiTools, generationConfig: gen }));
  return { c1, c2 };
}
async function turnVyce(model, focus, maxOut) {
  const messages = [{ role: 'system', content: SYSTEM_BIG }, { role: 'user', content: userBlock(focus) }];
  const c1 = await timed(() => vyce(model, { messages, tools: oaiTools, max_tokens: maxOut }));
  if (c1.error || c1.fc !== 'search_knowledge') return { c1, c2: null };
  const tc = c1.msg.tool_calls[0];
  const messages2 = [...messages, { role: 'assistant', content: c1.msg.content ?? '', tool_calls: c1.msg.tool_calls, ...(c1.msg.reasoning_content ? { reasoning_content: c1.msg.reasoning_content } : {}) },
    ...c1.msg.tool_calls.map((t) => ({ role: 'tool', tool_call_id: t.id, content: JSON.stringify(t.id === tc.id ? TOOL_RESULT : { ok: true }) }))];
  const c2 = await timed(() => vyce(model, { messages: messages2, tools: oaiTools, max_tokens: maxOut }));
  return { c1, c2 };
}

// (3) embedding
async function embedOnce(text) {
  if (calls >= MAX_CALLS) throw new Error('MAX_CALLS');
  calls++;
  const model = 'gemini-embedding-001';
  const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${model}:batchEmbedContents`, {
    method: 'POST', headers: { 'content-type': 'application/json', 'x-goog-api-key': GEMINI_KEY },
    body: JSON.stringify({ requests: [{ model: `models/${model}`, content: { parts: [{ text }] }, taskType: 'RETRIEVAL_QUERY', outputDimensionality: 1024 }] }), signal: AbortSignal.timeout(30_000),
  });
  const j = await res.json().catch(() => null);
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return { vec: j.embeddings[0].values, dim: j.embeddings[0].values.length };
}

function summarizeMeta(ms) {
  const ok = ms.filter((m) => !m.error && (m.finish !== undefined || m.dim !== undefined));
  if (ok.length === 0) return '';
  const fin = {}; const fc = {};
  for (const m of ok) { if (m.finish) fin[m.finish] = (fin[m.finish] ?? 0) + 1; if (m.fc !== undefined) fc[m.fc ?? 'text'] = (fc[m.fc ?? 'text'] ?? 0) + 1; }
  const avg = (k) => Math.round(ok.reduce((a, m) => a + (m[k] ?? 0), 0) / ok.length);
  if (ok[0].dim !== undefined) return `dim=${[...new Set(ok.map((m) => m.dim))].join(',')}`;
  return `finish=${JSON.stringify(fin)} out=${JSON.stringify(fc)} tb_in=${avg('inTok')} tb_out=${avg('outTok')} tb_think=${avg('thinkTok')}`;
}

      create table kb_sections (id serial primary key, heading text, body text not null,
        tsv tsvector generated always as (setweight(to_tsvector('simple', public.immutable_unaccent(coalesce(heading,''))), 'A') || to_tsvector('simple', public.immutable_unaccent(body))) stored,
        embedding vector(1024));
      create index on kb_sections using gin(tsv);`);
    const N = 3000;
    const words = ['agent', 'zalo', 'chi phí', 'setup', 'bot', 'khách hàng', 'kho tri thức', 'model', 'token', 'nhóm', 'kiểm duyệt', 'bảng giá', 'shop', 'đơn hàng', 'tự động', 'trả lời', 'tài liệu', 'cộng đồng', 'prompt', 'tool'];
    await c.query(`insert into kb_sections (heading, body, embedding)
      select 'Mục ' || g, (select string_agg((array[${words.map((w) => `'${w}'`).join(',')}])[1 + floor(random()*${words.length})::int], ' ') from generate_series(1, 120 + (g % 7)) x where g > 0),
             (select array_agg(random() - 0.5)::vector(1024) from generate_series(1,1024) y where g > 0)
      from generate_series(1, ${N}) g`);
    await c.query('analyze kb_sections');
    const cnt = (await c.query('select count(*)::int n, avg(length(body))::int l from kb_sections')).rows[0];
    log(`# DB tam ${DB}: kb_sections ${cnt.n} dong (tb ${cnt.l} ky tu), GIN tsv, vector(1024) KHONG index (dung DDL hien tai) · pgvector ${(await c.query("select extversion v from pg_extension where extname='vector'")).rows[0].v}`);
    const vec = `[${(qvec ?? Array.from({ length: 1024 }, () => Math.random() - 0.5)).join(',')}]`;
    const tsq = 'chi | phi | setup | bot | cskh | zalo | shop';
    const LEX = `select id, ts_rank_cd(tsv, to_tsquery('simple', $1), 32) r from kb_sections where tsv @@ to_tsquery('simple', $1) order by r desc limit 20`;
    const VEC = `select id from kb_sections where embedding is not null order by embedding <=> $1::vector limit 20`;
    const RRF = `with l as (select id, row_number() over (order by ts_rank_cd(tsv, to_tsquery('simple', $1), 32) desc) rk from kb_sections where tsv @@ to_tsquery('simple', $1) limit 20),
      v as (select id, row_number() over (order by embedding <=> $2::vector) rk from (select id, embedding from kb_sections where embedding is not null order by embedding <=> $2::vector limit 20) s)
      select id, sum(1.0/(60+rk)) s from (select * from l union all select * from v) u group by id order by s desc limit 8`;
    for (let i = 0; i < 3; i++) { await c.query('select 1'); await c.query(LEX, [tsq]); await c.query(VEC, [vec]); }
    for (let i = 0; i < 30; i++) {
      let t = performance.now(); await c.query('select 1'); rec('4.db roundtrip select 1', performance.now() - t);
      t = performance.now(); await c.query(LEX, [tsq]); rec('4.db lexical ts_rank_cd top20 (3000 muc)', performance.now() - t);
      t = performance.now(); await c.query(VEC, [vec]); rec('4.db vector <=> top20 seqscan (3000 muc)', performance.now() - t);
      t = performance.now(); await c.query(RRF, [tsq, vec]); rec('4.db lexical+vector+RRF 1 cau SQL', performance.now() - t);
    }
  } finally {
    await c.end();
    await admin.query(`drop database if exists ${DB}`);
    const left = (await admin.query('select count(*)::int n from pg_database where datname = $1', [DB])).rows[0].n;
    log(`# DB tam da xoa: con lai ${left} database ten ${DB}`);
    await admin.end();
  }
}
