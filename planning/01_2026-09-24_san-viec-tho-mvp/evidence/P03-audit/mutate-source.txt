// H05 audit: ap mot dot bien (mutation) vao file trong worktree. Dung: node mutate.mjs <file> <search-json> <replace-json>
import { readFileSync, writeFileSync } from 'node:fs';
const [file, s, r] = process.argv.slice(2);
const search = JSON.parse(s);
const replace = JSON.parse(r);
const src = readFileSync(file, 'utf8');
const n = src.split(search).length - 1;
if (n !== 1) {
  console.log(`MUTATE_FAIL: tim thay ${n} lan trong ${file}`);
  process.exit(2);
}
writeFileSync(file, src.replace(search, replace));
console.log(`MUTATED ${file}`);
