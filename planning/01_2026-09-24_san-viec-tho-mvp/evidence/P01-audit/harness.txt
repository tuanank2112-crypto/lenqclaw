// Audit H02 — independent, multi-line-aware scan of the audited tree for
// BB-04/BB-07 violations (to separate "checker blind spot" from "real violation").
import { execSync } from 'node:child_process';
import { readFileSync } from 'node:fs';
const root = process.env.WT;
const files = execSync('git ls-files apps packages scripts', { cwd: root, encoding: 'utf8' })
  .split('\n').filter((f) => /\.(ts|tsx|js|mjs|cjs|mts|cts|jsx)$/.test(f));
const ALLOW = {
  pg: ['packages/core/src/adapters/db.ts'],
  sdk: ['packages/core/src/adapters/model.ts'],
  zca: ['packages/core/src/adapters/transport.ts'],
  outbox: ['packages/core/src/adapters/transport.ts', 'scripts/check_adapter_boundary.mjs'],
};
// module specifiers inside import/require/export-from, across newlines, any quote style
const SPEC = /(?:\bfrom\s*|\bimport\s*\(\s*|\brequire\s*\(\s*|^\s*import\s+)(['"`])([^'"`]+)\1/gm;
const OUTBOX = /insert(?:\s|\/\*[\s\S]*?\*\/)+into\s+(?:"?public"?\.)?"?outbox\b/gi;
let n = 0;
const lineOf = (s, i) => s.slice(0, i).split('\n').length;
for (const f of files) {
  const s = readFileSync(`${root}/${f}`, 'utf8');
  for (const m of s.matchAll(SPEC)) {
    const spec = m[2];
    const kind = /^pg(\/|$)|^pg-pool$/.test(spec) ? 'pg' : /^@anthropic-ai\/sdk/.test(spec) ? 'sdk' : /^zca-js/.test(spec) ? 'zca' : null;
    if (!kind) continue;
    const typeOnly = /^\s*import\s+type\b/.test(s.slice(s.lastIndexOf('\n', m.index) + 1, m.index + m[0].length)) ;
    const allowed = ALLOW[kind].includes(f) || (kind === 'zca' && f.startsWith('apps/zalo-worker/'));
    console.log(`${allowed ? 'ok  ' : typeOnly ? 'type' : 'VIOL'} ${f}:${lineOf(s, m.index)} specifier=${spec}`);
    if (!allowed && !typeOnly) n++;
  }
  for (const m of s.matchAll(OUTBOX)) {
    const allowed = ALLOW.outbox.includes(f);
    console.log(`${allowed ? 'ok  ' : 'VIOL'} ${f}:${lineOf(s, m.index)} outbox-insert="${m[0].replace(/\s+/g, ' ')}"`);
    if (!allowed) n++;
  }
}
console.log(`files_scanned: ${files.length}`);
console.log(`real_violations: ${n}`);
