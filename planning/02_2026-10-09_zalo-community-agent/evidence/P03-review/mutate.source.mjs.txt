const id = process.argv[2];
const m = M[id];
if (!m) {
  console.error('unknown', id);
  process.exit(2);
}
const src = readFileSync(m.file, 'utf8').replace(/\r\n/g, '\n');
const n = src.split(m.from).length - 1;
if (n !== 1) {
  console.error(`MUTATION_NOT_APPLIED ${id}: khop ${n} lan`);
  process.exit(2);
}
writeFileSync(m.file, src.replace(m.from, m.to));
console.log(`=== ${id} [nhom ${m.group}] ${m.file}: ${m.why}`);
const results = [];
try {
  for (const r of m.run) {
    const p = spawnSync(r.cmd, { cwd: r.cwd, shell: true, encoding: 'utf8' });
    const out = (p.stdout + p.stderr).replace(/\x1b\[[0-9;]*m/g, '');
    const tail = out
      .split('\n')
      .filter((l) => /violations:|Tests |FAIL|✗|×|AssertionError|expected|Error:|node "|bi cam/.test(l))
      .slice(0, 12)
      .join('\n');
    console.log(`$ (${r.cwd}) ${r.cmd}\nexit=${p.status}\n${tail}`);
    results.push(p.status);
  }
} finally {
  const co = spawnSync(`git checkout -- "${m.file}"`, { shell: true, encoding: 'utf8' });
  const st = spawnSync(`git status --short -- "${m.file}"`, { shell: true, encoding: 'utf8' });
  console.log(`restore: git checkout exit=${co.status}; status="${st.stdout.trim()}"`);
}
const caught = results.some((c) => c !== 0);
console.log(`VERDICT ${id}: ${caught ? 'BAT DUOC (do)' : 'LOT (xanh)'}`);
